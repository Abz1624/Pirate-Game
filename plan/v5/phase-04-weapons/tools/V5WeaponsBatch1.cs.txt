using System.Collections.Generic;
using System.Linq;
using RoguePirates.V4.Config;
using RoguePirates.V4.Presentation.Hud;
using UnityEditor;
using UnityEngine;

namespace RoguePirates.V4.Editor
{
    /// <summary>
    /// V5 M4: writes the first batch of new weapons (Anchor Flail, Chainshot, Siren Bell, Ramming Prow), the eight
    /// Empowered evolutions (the four new weapons, Harpoon Gun, Long Nine, Mortar, Swivel Guns) and their eight chest
    /// items, and registers them (config set, unlock prices, stage profile, icons). Each asset starts as a copy of an
    /// existing one, so every field it does not set keeps a working value. Idempotent: running it again rewrites the same
    /// values. Damage is set against today's weapons of the same role (plan/v5/phase-04-weapons/PLAN.md, D31).
    /// </summary>
    public static class V5WeaponsBatch1
    {
        const string SetPath = "Assets/_Game/V4/Data/ConfigSets/SaltwindDefault.asset";
        const string WeaponDir = "Assets/_Game/V4/Data/Weapons/";
        const string EvolutionDir = "Assets/_Game/V4/Data/Evolutions/";
        const string ItemDir = "Assets/_Game/V4/Data/Build/Items/";
        const string BalanceVersion = "saltwind-v5-m4";

        /// <summary>The four new weapons and their unlock prices in Rum (D31).</summary>
        public static readonly (string id, int rum)[] Weapons =
        {
            ("anchor_flail", 120), ("chainshot", 130), ("siren_bell", 150), ("ramming_prow", 160),
        };

        [MenuItem("RoguePirates/V5/Write M4 weapons batch 1")]
        public static string Apply()
        {
            var set = AssetDatabase.LoadAssetAtPath<ConfigSet>(SetPath);
            if (set == null) return "config set not found";
            var nextOrder = set.All<WeaponConfig>().Where(w => !Weapons.Any(n => n.id == w.weaponId)).Max(w => w.databaseOrder) + 1;
            var cannonball = AssetDatabase.LoadAssetAtPath<GameObject>("Assets/GameObject/Cannonball.prefab");

            var flail = Weapon("anchor_flail", "flying_cutlasses", w =>
            {
                w.displayName = "Anchor Flail";
                w.baseCooldown = 0.75f; // the per-target interval follows the cooldown; the model reads damage / cooldown
                w.description = "A heavy anchor swings around the ship, battering and shoving foes.";
                w.behaviorLine = "Swings around the ship";
                w.baseDamage = 12f;
                w.range = 6f;
                w.procCoefficient = 0.6f;
                w.orbit.baseBlades = 1;
                w.orbit.orbitRadius = 3.5f;
                w.orbit.angularSpeedDegrees = 100f;
                w.orbit.hitRadius = 1.6f;
                w.orbit.damageMultiplier = 1f;
                w.orbit.perTargetIntervalSeconds = 0.75f;
                w.orbit.knockbackMeters = 2f;
                w.visual.orbitModel = cannonball; // placeholder until the anchor model (art guide)
                w.visual.orbitModelScale = 1.3f;
                AuraAxes(w, ConfigEnums.UpgradeAxis.Knockback, 2.2f); // one anchor; the second comes with Kraken Chain
                w.aimOverlayColor = new Color(0.6f, 0.62f, 0.7f);
            }, ref nextOrder);
            var chain = Weapon("chainshot", "deck_cannons", w =>
            {
                w.displayName = "Chainshot";
                w.description = "Spinning chained balls that tear through a line of foes and slow them.";
                w.behaviorLine = "Fires forward, passes through";
                w.baseDamage = 9f;
                w.baseCooldown = 1.5f;
                w.range = 16f;
                w.projectileSpeed = 18f;
                w.projectile.sizeBase = 0.75f;
                w.projectile.lifeSeconds = 1f;
                w.projectile.homing = false;
                w.projectile.pierce = 2;
                w.projectile.slowFraction = 0.25f;
                w.projectile.slowSeconds = 2f;
                // Projectiles do not knock back: no Knockback card (Deck Cannons keep theirs, question 17).
                w.upgradeAxes = w.upgradeAxes.Where(a => a.axis != ConfigEnums.UpgradeAxis.Knockback).ToArray();
                w.validUpgradeAxes = w.upgradeAxes.Select(a => a.axis).ToArray();
                w.aimOverlayColor = new Color(0.85f, 0.85f, 0.92f);
            }, ref nextOrder);
            var bell = Weapon("siren_bell", "lightning_mast", w =>
            {
                w.displayName = "Siren Bell";
                w.description = "A cursed bell tolls around the ship, hurting and slowing everything near.";
                w.behaviorLine = "Pulses around the ship";
                w.pattern = ConfigEnums.WeaponPattern.AuraPulse;
                w.behavior = ConfigEnums.WeaponBehavior.Aura;
                w.targeting = ConfigEnums.TargetingMode.AreaAroundShip;
                w.baseDamage = 12f;
                w.baseCooldown = 3f;
                w.range = 5f;
                w.procCoefficient = 0.5f;
                w.aura.radiusBase = 5f;
                w.aura.damageMultiplier = 1f;
                w.aura.curseSeconds = 0f;
                w.aura.slowFraction = 0.2f;
                w.aura.slowSeconds = 1.5f;
                w.aura.requiresBoost = false;
                w.visual.aimShape = ConfigEnums.AimShape.ByTargeting; w.visual.firePulseSeconds = 0.18f; w.visual.firePulseFades = false;
                w.aura.forwardOffset = 0f;
                w.aura.knockbackMeters = 0f;
                AuraAxes(w, null, 1.6f); // Duration would only lengthen the slow
                w.aimOverlayColor = new Color(0.45f, 0.95f, 0.95f);
            }, ref nextOrder);
            var prow = Weapon("ramming_prow", "lightning_mast", w =>
            {
                w.displayName = "Ramming Prow";
                w.description = "An iron prow: while you boost, it smashes and shoves everything ahead.";
                w.behaviorLine = "Strikes ahead while boosting";
                w.pattern = ConfigEnums.WeaponPattern.AuraPulse;
                w.behavior = ConfigEnums.WeaponBehavior.Aura;
                w.targeting = ConfigEnums.TargetingMode.AreaAroundShip;
                w.baseDamage = 14f;
                w.baseCooldown = 0.3f;
                w.range = 4f;
                w.procCoefficient = 0.4f;
                w.aura.radiusBase = 3f;
                w.aura.damageMultiplier = 1f;
                w.aura.curseSeconds = 0f;
                w.aura.slowFraction = 0f;
                w.aura.slowSeconds = 0f;
                w.aura.requiresBoost = true;
                w.visual.aimShape = ConfigEnums.AimShape.ByTargeting; w.visual.firePulseSeconds = 0.18f; w.visual.firePulseFades = false;
                w.aura.forwardOffset = 2.5f;
                w.aura.knockbackMeters = 2.5f;
                AuraAxes(w, ConfigEnums.UpgradeAxis.Knockback, 2.2f);
                w.aimOverlayColor = new Color(0.95f, 0.75f, 0.35f);
            }, ref nextOrder);

            var items = new List<ItemConfig>();
            var evolutions = new List<WeaponEvolutionConfig>();
            var itemOrder = set.All<ItemConfig>().Where(i => !IsNewItem(i.itemId)).Max(i => i.chestPoolOrder) + 1;
            var collection = set.All<ItemConfig>().Where(i => !IsNewItem(i.itemId)).Max(i => i.collectionOrder) + 1;

            void Pair(string evoId, string evoName, string evoText, WeaponConfig weapon, string itemId, string itemName, int stat, float add, float mult, string statText, System.Action<EmpowerBlock> empower)
            {
                var item = Copy<ItemConfig>(ItemDir, "grapeshot_pouch", itemId);
                item.configId = "item." + itemId;
                item.itemId = itemId;
                item.displayName = itemName;
                item.description = $"{statText}; evolves {weapon.displayName} into {evoName}.";
                item.rarity = 1;
                item.inChestPool = true;
                item.inLevelUpPool = false;
                item.chestPoolOrder = itemOrder++;
                item.collectionOrder = collection++;
                item.statModifiers = new[] { new ItemStatModifier { stat = stat, additive = add, multiplicative = mult } };
                item.effects = new ItemEffect[0];
                item.balanceVersion = BalanceVersion;
                Save(item);
                items.Add(item);

                var evo = Copy<WeaponEvolutionConfig>(EvolutionDir, "stormshot_cannon", evoId);
                evo.configId = "evolution." + evoId;
                evo.evolutionId = evoId;
                evo.displayName = evoName;
                evo.description = evoText;
                evo.baseWeapon = weapon;
                evo.requiredItemId = itemId;
                evo.effect = ConfigEnums.EvolutionEffect.Empowered;
                evo.empower = new EmpowerBlock();
                empower(evo.empower);
                evo.balanceVersion = BalanceVersion;
                Save(evo);
                evolutions.Add(evo);
            }

            WeaponConfig Existing(string id) => set.All<WeaponConfig>().First(w => w.weaponId == id);

            Pair("kraken_chain", "Kraken Chain", "A second anchor; both swing wider and hit harder.", flail,
                "rusted_shackle", "Rusted Shackle", 10, 0f, 1.12f, "+12% size",
                e => { e.extraQuantity = 1; e.damageMultiplier = 1.3f; e.sizeMultiplier = 1.25f; e.knockbackMultiplier = 1.25f; });
            Pair("bar_shot_barrage", "Bar Shot Barrage", "Three chained shots that pass through two more foes.", chain,
                "chain_locker", "Chain Locker", 12, 0f, 1.1f, "+10% projectile speed",
                e => { e.extraQuantity = 2; e.extraPierce = 2; e.damageMultiplier = 1.2f; });
            Pair("drowned_choir", "Drowned Choir", "The bell tolls faster and wider, and slows harder.", bell,
                "siren_scale", "Siren Scale", 6, 0f, 1.09f, "+9% attack speed",
                e => { e.sizeMultiplier = 1.4f; e.cooldownMultiplier = 0.75f; e.extraSlow = 0.1f; e.damageMultiplier = 1.25f; });
            Pair("ironclad_ram", "Ironclad Ram", "A heavier prow: wider, harder strikes while boosting.", prow,
                "iron_figurehead", "Iron Figurehead", 5, 0f, 1.09f, "+9% damage",
                e => { e.damageMultiplier = 1.6f; e.sizeMultiplier = 1.35f; e.knockbackMultiplier = 1.3f; e.cooldownMultiplier = 0.8f; });
            Pair("whalers_volley", "Whaler's Volley", "Two harpoons at once that pass through a foe.", Existing("harpoon_gun"),
                "whale_oil_barrel", "Whale Oil Barrel", 5, 0f, 1.09f, "+9% damage",
                e => { e.extraQuantity = 1; e.extraPierce = 1; e.damageMultiplier = 1.25f; });
            Pair("ship_of_the_line", "Ship of the Line", "A wider, heavier beam that fires more often.", Existing("long_nine"),
                "powder_horn_of_the_line", "Powder Horn of the Line", 5, 0f, 1.09f, "+9% damage",
                e => { e.damageMultiplier = 1.5f; e.sizeMultiplier = 1.5f; e.cooldownMultiplier = 0.85f; });
            Pair("siege_bombard", "Siege Bombard", "Bigger, harder shells, more often.", Existing("mortar"),
                "siege_quadrant", "Siege Quadrant", 10, 0f, 1.12f, "+12% size",
                e => { e.damageMultiplier = 1.35f; e.sizeMultiplier = 1.35f; e.cooldownMultiplier = 0.85f; });
            Pair("swivel_battery", "Swivel Battery", "Two more swivel shots, faster.", Existing("swivel_guns"),
                "swivel_mount", "Swivel Mount", 6, 0f, 1.09f, "+9% attack speed",
                e => { e.extraQuantity = 2; e.cooldownMultiplier = 0.8f; e.damageMultiplier = 1.15f; });

            // The config set.
            Undo.RecordObject(set, "V5 M4 weapons");
            foreach (var c in new V4ConfigAsset[] { flail, chain, bell, prow }.Concat(evolutions).Concat(items))
                if (!set.configs.Contains(c)) set.configs.Add(c);
            EditorUtility.SetDirty(set);

            // Unlock prices (the unlock config and the stage profile mirror each other).
            var unlocks = set.All<UnlockConfig>().First();
            var list = unlocks.weaponUnlocks.Where(u => !Weapons.Any(n => n.id == u.weaponId)).ToList();
            list.AddRange(Weapons.Select(n => new WeaponUnlock { weaponId = n.id, rumCost = n.rum }));
            unlocks.weaponUnlocks = list.ToArray();
            EditorUtility.SetDirty(unlocks);
            foreach (var stage in set.All<StageContentConfig>())
            {
                if (stage.sharedProfile == null) continue;
                var shared = stage.sharedProfile.weaponUnlocks.Where(u => !Weapons.Any(n => n.id == u.contentId)).ToList();
                shared.AddRange(Weapons.Select(n => new SharedUnlockEntry { contentId = n.id, marksCost = n.rum }));
                stage.sharedProfile.weaponUnlocks = shared.ToArray();
                EditorUtility.SetDirty(stage);
            }

            // Icons: placeholders from the nearest existing weapon or item until the art guide's icons arrive.
            var icons = Resources.Load<IconCatalog>(IconCatalog.ResourcePath);
            if (icons != null)
            {
                Icon(icons.weapons, "anchor_flail", icons.Weapon("flying_cutlasses"));
                Icon(icons.weapons, "chainshot", icons.Weapon("deck_cannons"));
                Icon(icons.weapons, "siren_bell", icons.Weapon("lightning_mast"));
                Icon(icons.weapons, "ramming_prow", icons.Weapon("broadside_battery"));
                foreach (var item in items) Icon(icons.items, item.itemId, icons.Item("grapeshot_pouch"));
                EditorUtility.SetDirty(icons);
            }

            AssetDatabase.SaveAssets();
            return $"ok: 4 weapons, {evolutions.Count} evolutions, {items.Count} items";
        }

        /// <summary>Tunes the four new weapons' base damage and growth with the balance model against the weapons a regular
        /// player owns (held fixed), as M3 tuned those, and writes them.</summary>
        [MenuItem("RoguePirates/V5/Tune M4 weapons batch 1")]
        public static string Tune()
        {
            var set = AssetDatabase.LoadAssetAtPath<ConfigSet>(SetPath);
            using var rt = new ConfigSetRuntime(set);
            var content = new RoguePirates.V4.DevTools.Balance.SimContent(rt) { RegularPlayerRum = int.MaxValue };
            var ids = Weapons.Select(w => w.id).ToList();
            var tunes = RoguePirates.V4.DevTools.Balance.WeaponModel.TuneOnly(content, ids);
            var report = new System.Text.StringBuilder();
            foreach (var t in tunes)
            {
                var tuned = rt.All<WeaponConfig>().First(w => w.weaponId == t.WeaponId);
                var asset = set.All<WeaponConfig>().First(w => w.weaponId == t.WeaponId);
                Undo.RecordObject(asset, "V5 M4 tune");
                asset.baseDamage = tuned.baseDamage;
                for (var i = 0; i < asset.upgradeAxes.Length && i < tuned.upgradeAxes.Length; i++)
                {
                    asset.upgradeAxes[i].baseIncrement = tuned.upgradeAxes[i].baseIncrement;
                    asset.upgradeAxes[i].maxAccumulated = tuned.upgradeAxes[i].maxAccumulated;
                }
                EditorUtility.SetDirty(asset);
                report.AppendLine($"{t.WeaponId}: damage x{t.DamageScale:0.000} (now {asset.baseDamage:0.00}), growth x{t.GrowthScale:0.00}, worst {t.WorstBefore:0.00} -> {t.WorstAfter:0.00}");
            }
            AssetDatabase.SaveAssets();
            System.IO.File.WriteAllText("../plan/v5/phase-04-weapons/m4/tune.txt", report.ToString());
            return report.ToString();
        }

        static readonly string[] NewItems =
        {
            "rusted_shackle", "chain_locker", "siren_scale", "iron_figurehead", "whale_oil_barrel", "powder_horn_of_the_line",
            "siege_quadrant", "swivel_mount",
        };

        public static bool IsNewItem(string itemId) => NewItems.Contains(itemId);

        static WeaponConfig Weapon(string id, string from, System.Action<WeaponConfig> set, ref int nextOrder)
        {
            var w = Copy<WeaponConfig>(WeaponDir, from, id);
            var order = w.weaponId == id && w.databaseOrder >= 0 ? w.databaseOrder : nextOrder++;
            w.configId = "weapon." + id;
            w.weaponId = id;
            w.databaseOrder = order;
            w.tags = new string[0];
            w.levelDamageFactor = true;
            w.runtimeCatalogWeapon = false;
            set(w);
            w.balanceVersion = BalanceVersion;
            Save(w);
            return w;
        }

        /// <summary>Level-up lines without quantity (a pulse has none; the Flail's second anchor is its evolution): the
        /// Lightning Mast's damage and cooldown, then size and one more, increments × growth.</summary>
        static void AuraAxes(WeaponConfig w, ConfigEnums.UpgradeAxis? last, float growth)
        {
            var src = AssetDatabase.LoadAssetAtPath<WeaponConfig>(WeaponDir + "lightning_mast.asset");
            WeaponUpgradeAxis Of(ConfigEnums.UpgradeAxis a) => src.upgradeAxes.First(x => x.axis == a);
            WeaponUpgradeAxis Scaled(ConfigEnums.UpgradeAxis a, float weight) =>
                new WeaponUpgradeAxis { axis = a, baseIncrement = 0.034827992f * growth, offerWeight = weight, maxAccumulated = 0.8955768f * growth };
            var damage = Of(ConfigEnums.UpgradeAxis.Damage);
            var cooldown = Of(ConfigEnums.UpgradeAxis.Cooldown);
            w.upgradeAxes = new[]
            {
                new WeaponUpgradeAxis { axis = damage.axis, baseIncrement = damage.baseIncrement * growth, offerWeight = damage.offerWeight, maxAccumulated = damage.maxAccumulated * growth },
                new WeaponUpgradeAxis { axis = cooldown.axis, baseIncrement = cooldown.baseIncrement, offerWeight = cooldown.offerWeight, maxAccumulated = cooldown.maxAccumulated },
                Scaled(ConfigEnums.UpgradeAxis.Size, 0.9f),
            };
            if (last.HasValue) w.upgradeAxes = w.upgradeAxes.Append(Scaled(last.Value, 0.8f)).ToArray();
            w.validUpgradeAxes = w.upgradeAxes.Select(a => a.axis).ToArray();
        }

        static T Copy<T>(string dir, string from, string to) where T : ScriptableObject
        {
            var path = dir + to + ".asset";
            var existing = AssetDatabase.LoadAssetAtPath<T>(path);
            if (existing != null) return existing;
            if (!AssetDatabase.CopyAsset(dir + from + ".asset", path)) throw new System.InvalidOperationException("copy failed: " + path);
            AssetDatabase.ImportAsset(path);
            return AssetDatabase.LoadAssetAtPath<T>(path);
        }

        static void Save(Object o) => EditorUtility.SetDirty(o);

        static void Icon(List<IconCatalog.Entry> list, string id, Sprite sprite)
        {
            list.RemoveAll(e => e.id == id);
            list.Add(new IconCatalog.Entry { id = id, sprite = sprite });
        }
    }
