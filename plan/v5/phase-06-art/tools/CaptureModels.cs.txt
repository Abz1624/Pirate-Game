var log = new System.Text.StringBuilder();
var pru = new UnityEditor.PreviewRenderUtility();
try {
  var cam = pru.camera;
  cam.fieldOfView = 30f;
  cam.clearFlags = CameraClearFlags.SolidColor;
  cam.backgroundColor = new Color(0.87f, 0.91f, 0.94f, 1f);
  cam.nearClipPlane = 0.01f; cam.farClipPlane = 5000f;
  pru.lights[0].intensity = 1.25f; pru.lights[0].transform.rotation = Quaternion.Euler(50f, 210f, 0f);
  pru.lights[1].intensity = 0.55f; pru.lights[1].transform.rotation = Quaternion.Euler(20f, 40f, 0f);
  pru.ambientColor = new Color(0.45f, 0.47f, 0.5f);
  foreach (var (slug, path) in assets) {
    var src = UnityEditor.AssetDatabase.LoadAssetAtPath<GameObject>(path);
    if (src == null) { log.AppendLine($"MISSING\t{slug}\t{path}"); continue; }
    var go = pru.InstantiatePrefabInScene(src);
    go.transform.position = Vector3.zero;
    foreach (var r in go.GetComponentsInChildren<Renderer>(true))
      if (!(r is MeshRenderer || r is SkinnedMeshRenderer)) r.enabled = false;
    foreach (var c in go.GetComponentsInChildren<Canvas>(true)) c.enabled = false;
    foreach (var lod in go.GetComponentsInChildren<LODGroup>(true)) {
      var lods = lod.GetLODs();
      lod.enabled = false;
      for (int i = 1; i < lods.Length; i++) foreach (var r in lods[i].renderers) if (r) r.enabled = false;
    }
    var rends = System.Linq.Enumerable.ToArray(System.Linq.Enumerable.Where(go.GetComponentsInChildren<Renderer>(true),r => r.enabled && r.gameObject.activeInHierarchy && (r is MeshRenderer || r is SkinnedMeshRenderer)));
    if (rends.Length == 0) { log.AppendLine($"NOMESH\t{slug}\t{path}"); UnityEngine.Object.DestroyImmediate(go); continue; }
    var b = rends[0].bounds; foreach (var r in rends) b.Encapsulate(r.bounds);
    long tris = 0;
    foreach (var r in rends) {
      Mesh m = r is SkinnedMeshRenderer sk ? sk.sharedMesh : r.GetComponent<MeshFilter>()?.sharedMesh;
      if (m != null) for (int s = 0; s < m.subMeshCount; s++) tris += m.GetIndexCount(s) / 3;
    }
    var radius = b.extents.magnitude;
    var rot = Quaternion.Euler(40f, 225f, 0f);
    var dist = radius / Mathf.Sin(cam.fieldOfView * 0.5f * Mathf.Deg2Rad) * 1.05f;
    cam.transform.rotation = rot;
    cam.transform.position = b.center - rot * Vector3.forward * dist;
    pru.BeginStaticPreview(new Rect(0, 0, 512, 512));
    pru.Render(true);
    var tex = pru.EndStaticPreview();
    System.IO.File.WriteAllBytes(System.IO.Path.Combine(outDir, slug + ".png"), tex.EncodeToPNG());
    UnityEngine.Object.DestroyImmediate(tex);
    UnityEngine.Object.DestroyImmediate(go);
    log.AppendLine($"OK\t{slug}\t{tris}\t{b.size.x:F1}x{b.size.y:F1}x{b.size.z:F1}\t{path}");
  }
} finally { pru.Cleanup(); }
System.IO.File.WriteAllText(System.IO.Path.Combine(outDir, "renders.tsv"), log.ToString());
return log.ToString();
